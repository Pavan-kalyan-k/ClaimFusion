    for dirpath, dirnames, _ in os.walk(root):

import numpy as np
import seaborn as sns
from sklearn.metrics import classification_report, confusion_matrix

# FIX: these two plotting/reporting blocks were copy-pasted 4-6 times
# throughout the original notebook (once per model / fine-tuning
# stage) with only cosmetic differences. Pulling them into functions
# here means each model's results are produced by a single call below
# instead of a duplicated block.

def evaluate_model(model, generator, title):
    """Print loss/accuracy, confusion matrix, and classification report."""
    loss, acc = model.evaluate(generator, verbose=0)
    print(f"{title} -- Validation loss: {loss:.4f}, accuracy: {acc:.4f}")

    generator.reset()
    y_true = generator.classes
    y_pred_probs = model.predict(generator, verbose=0)
    y_pred = np.argmax(y_pred_probs, axis=1)
    labels = list(generator.class_indices.keys())

    cm = confusion_matrix(y_true, y_pred)
    plt.figure(figsize=(6, 5))
    sns.heatmap(cm, annot=True, fmt="d", cmap="Blues", xticklabels=labels, yticklabels=labels)
    plt.title(f"Confusion Matrix -- {title}")
    plt.xlabel("Predicted Label")
    plt.ylabel("True Label")
    plt.show()

    print(f"\nClassification report -- {title}\n")
    print(classification_report(y_true, y_pred, target_names=labels))
    return acc


def show_sample_predictions(model, generator, num_samples=9):
    """Show a grid of random predictions vs. actual labels."""
    generator.reset()
    x_batch, y_batch_true = next(generator)
    preds = model.predict(x_batch, verbose=0)
    labels_map = {v: k for k, v in generator.class_indices.items()}

    n = min(num_samples, len(x_batch))
    fig, axes = plt.subplots(3, 3, figsize=(12, 12))
    axes = axes.ravel()
    for i in range(n):
        actual_label = labels_map[np.argmax(y_batch_true[i])]
        pred_label = labels_map[np.argmax(preds[i])]
        confidence = np.max(preds[i]) * 100

        axes[i].imshow(x_batch[i])
        axes[i].axis("off")
        color = "green" if actual_label == pred_label else "red"
        axes[i].set_title(
            f"Actual: {actual_label}\nPred: {pred_label} ({confidence:.1f}%)",
            color=color, fontsize=10, fontweight="bold",
        )
    for j in range(n, len(axes)):
        axes[j].axis("off")
    plt.tight_layout()
    plt.show()

def calculate_iou(box1, box2):

    x1 = max(box1[0], box2[0])
    y1 = max(box1[1], box2[1])

    x2 = min(box1[2], box2[2])
    y2 = min(box1[3], box2[3])

    intersection_width = max(0, x2 - x1)
    intersection_height = max(0, y2 - y1)

    intersection_area = (
        intersection_width * intersection_height
    )

    area1 = (
        (box1[2] - box1[0]) *
        (box1[3] - box1[1])
    )

    area2 = (
        (box2[2] - box2[0]) *
        (box2[3] - box2[1])
    )

    union_area = area1 + area2 - intersection_area

    if union_area == 0:
        return 0

    return intersection_area / union_area


# Store detections
detections = []

if boxes is not None and len(boxes) > 0:

    for box in boxes:

        class_id = int(box.cls[0])
        class_name = detector.names[class_id]

        confidence = float(box.conf[0])

        x1, y1, x2, y2 = box.xyxy[0].tolist()

        detections.append({
            "class_id": class_id,
            "class_name": class_name,
            "confidence": confidence,
            "box": [x1, y1, x2, y2]
        })


# ============================================================

# ============================================================

detections = sorted(
    detections,
    key=lambda x: x["confidence"],
    reverse=True
)

for detection in detections:

    keep = True

                detection["box"],
                selected["box"]
            )

            # If heavily overlapping, consider duplicate
            if iou > IOU_THRESHOLD:

                break

        f"Image not found at: {image_path}"
    )

# Use low confidence so that valid parts are not removed

results = detector.predict(

result = results[0]

# ============================================================
# 5. IoU FUNCTION
# ============================================================

def calculate_iou(box1, box2):

    # Coordinates of intersection
    x1 = max(box1[0], box2[0])
    y1 = max(box1[1], box2[1])

    x2 = min(box1[2], box2[2])
    y2 = min(box1[3], box2[3])

    # Intersection dimensions
    intersection_width = max(0, x2 - x1)
    intersection_height = max(0, y2 - y1)

    intersection_area = (
        intersection_width * intersection_height
    )

    # Area of first box
    area1 = (
        max(0, box1[2] - box1[0]) *
        max(0, box1[3] - box1[1])
    )

    # Area of second box
    area2 = (
        max(0, box2[2] - box2[0]) *
        max(0, box2[3] - box2[1])
    )

    # Union
    union_area = area1 + area2 - intersection_area

    if union_area == 0:
        return 0

    return intersection_area / union_area

detections = []

    for box in boxes:

        class_id = int(box.cls[0])

        detections.append({

            "confidence": confidence,
            "box": [x1, y1, x2, y2]
        })

            f"{x2:.0f}, {y2:.0f})"

detections = sorted(
    detections,
    key=lambda x: x["confidence"],
    reverse=True
)

# ============================================================

# Higher value = more overlap required before
# boxes are considered duplicates.

IOU_THRESHOLD = 0.30

final_detections = []


for detection in detections:

    keep = True

    for selected in final_detections:

        # Compare only same classes
        if detection["class_id"] == selected["class_id"]:

            iou = calculate_iou(
                detection["box"],
                selected["box"]
            )

            # Same class + high overlap = duplicate
            if iou > IOU_THRESHOLD:

                keep = False

                break

    if keep:

        final_detections.append(detection)
